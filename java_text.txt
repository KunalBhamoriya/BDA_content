package assignment3;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;



public class SongManager{
    List<Song> songList = new ArrayList<>();

    public void addSong(Song s){
        songList.add(s);
        System.out.println("Song added successfully");
    }

    public void removeSongById(int serialNo){
        if(songList.removeIf(song -> song.getSerialNo() == serialNo)){
            System.out.println("Removed song with serial number " + serialNo);
        }else{
            System.out.println("No song found with serial number " + serialNo);
        }
    }
    
    public void updateSongTitle(int serialNo, String newTitle){
        for(Song song:songList){
            if(song.getSerialNo() == serialNo){
                song.setTitle(newTitle);
                System.out.println("Updated song title successfully");
                return;
            }
        }
        System.out.println("No song found with serial number " + serialNo);
    }


    public void updateRating(int serialNo, double newRating){
        songList.forEach(song->{
            if(song.getSerialNo() == serialNo){
                song.setRating(newRating);
                System.out.println("Updated song rating successfully");
                return;
            }
        });
    }

    public void displaySongs(){
        for(Song song:songList){
            System.out.println(song);
        }
    }


    // --------- 2 ----------

    public void sortSongsByTitle(){
        songList.sort(Comparator.comparing(song -> song.getTitle()));
    }


    //-------- 3 --------
    public Song searchSongByTitle(String title){
        for(Song song:songList){
            if(song.getTitle() == title){
                return song;
            }
        }
        return null;
    }


    //--------- 4 ----------

    public List<Song> getSongsByCurrentYear(){
        int currYear = java.time.Year.now().getValue();
        List<Song> recentSongs = new ArrayList<>();
        songList.forEach(song->{
            if(song.getYear() == currYear){
                recentSongs.add(song);
            }
        });
        return recentSongs;
    }
    

    // ---------- 5 -----------

    public List<Song> getSongsByArtist(String artistName){
        List<Song> temp = new ArrayList<>();
        for(Song s:songList){
            if(s.getArtists().contains(artistName)){
                temp.add(s);
            }
        }
        return temp;
    }

    // ----------6- -------

    public Map<String, Integer> getTitleYearList(){
        Map<String, Integer> map = new HashMap<>();
        songList.forEach(s->{
            map.put(s.getTitle(), s.getYear());
        });
        return map;
    }

    

    // ----------- 7 ----------
    public void DisplayLowestRatingSongs(){
        if(songList.isEmpty()){
            System.out.println("No songs in the list");
            return;
        }
        double minRating = songList.get(0).getRating();
        for(Song s:songList){
            if(s.getRating() < minRating){
                minRating = s.getRating();
            }
        }
        for(Song s:songList){
            if(s.getRating() == minRating){
                System.out.println(s);
            }
        }
    }

    public void DisplayHighestRatingSongs(){
        if(songList.isEmpty()){
            System.out.println("No songs in the list");
            return;
        }
        double maxRating = songList.get(0).getRating();
        for(Song s:songList){
            if(s.getRating() > maxRating){
                maxRating = s.getRating();
            }
        }
        for(Song s:songList){
            if(s.getRating() == maxRating){
                System.out.println(s);
            }
        }
    }
    
}

package assignment3;
import java.util.Set;

class Song{
    private int serialNo;
    private String title;
    private Set<String> artists;
    private int year;
    private double rating;

    public Song(int serialNo,String title, Set<String> artists,int year, double rating){
        this.serialNo=serialNo;
        this.title=title;
        this.artists=artists;
        this.year=year;
        this.rating=rating;
    }

    // getters and setters
    public int getSerialNo(){
        return serialNo;
    }

    public void setSerialNo(int serialNo){
        this.serialNo=serialNo;
    }
    
    public String getTitle(){
        return title;
    }

    public void setTitle(String title){
        this.title=title;
    }
    
    public Set<String> getArtists(){
        return artists;
    }
    public void setArtists(Set<String> artists){
        this.artists=artists;
    }
    public int getYear(){
        return year;
    }
    public void setYear(int year){
        this.year=year;
    }
    public double getRating(){
        return rating;
    }
    public void setRating(double rating){
        this.rating=rating;
    }
  

    @Override
    public String toString(){
        return "Song [serialNo=" + serialNo + ", title=" + title + ", artists=" + artists + ", year=" + year + ", rating=" + rating + "]";
    }
    
}

package assignment2;

public abstract class Vehicle implements InsurancePolicy {
    protected int registration_no;
    protected String brand;
    protected double rent;

    public void vehicle(int rno, String brand) {
        this.registration_no = rno;
        this.brand = brand;
    }

    public abstract double calculateRent(int hours);

    public int getRegistrationNo() {
        return registration_no;
    }

    public String getbrand() {
        return brand;
    }
}


package assignment2;

public class Bus extends Vehicle {
    private int seatCapacity;

    public Bus(int registrationNo, String brand, int seatCapacity) {
        this.registration_no = registrationNo;
        this.brand = brand;
        this.seatCapacity = seatCapacity;
    }

    @Override
    public double calculateRent(int hours) {
        if (hours <= 0)
            return 0.0;
        if (seatCapacity < 12) {
            return 50 * hours;
        } else {
            return 70 * hours;
        }
    }

    public int getSeatCapacity() {

        return seatCapacity;
    }

    public void setSeatCapacity(int seatCapacity) {
        this.seatCapacity = seatCapacity;
    }

    @Override
    public int calculatePremium() {
        if (seatCapacity > 12) {
            return 18000;
        } else {
            return 10000;
        }
    }
}
